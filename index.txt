        function simulateVehiclePhysics(carObj, upKey, downKey, leftKey, rightKey) {
            if (!carObj) return;
            
            // Allow driving outside active match if no match is currently running or ending
            if (!gameActive && (countdownActive || gameEnding)) {
                carObj.speed = 0;
                carObj.vx = 0;
                carObj.vy = 0;
                return;
            }

            if (upKey) {
                carObj.speed = Math.min(carObj.speed + carObj.accel, carObj.maxSpeed);
            } else if (downKey) {
                carObj.speed = Math.max(carObj.speed - carObj.brake, carObj.maxRev);
            } else {
                if (carObj.speed > 0) carObj.speed = Math.max(carObj.speed - carObj.friction, 0);
                if (carObj.speed < 0) carObj.speed = Math.min(carObj.speed + carObj.friction, 0);
            }

            if (leftKey) {
                carObj.angle -= carObj.maxSteer;
            } else if (rightKey) {
                carObj.angle += carObj.maxSteer;
            }

            let forwardX = Math.cos(carObj.angle), forwardY = Math.sin(carObj.angle);
            let slip = driftMode ? carObj.driftFactor : carObj.normalFactor;
            
            carObj.vx = carObj.vx * slip + (forwardX * carObj.speed) * (1 - slip);
            carObj.vy = carObj.vy * slip + (forwardY * carObj.speed) * (1 - slip);
            carObj.x += carObj.vx; 
            carObj.y += carObj.vy;

            if (driftMode && Math.abs(carObj.speed) > 3 && (leftKey || rightKey)) {
                skidmarks.push({x: carObj.x, y: carObj.y});
            }
        }

        function updatePhysics() {
            const p1Up = keys['w'] || keys['W'];
            const p1Down = keys['s'] || keys['S'];
            const p1Left = keys['a'] || keys['A'];
            const p1Right = keys['d'] || keys['D'];
            simulateVehiclePhysics(car, p1Up, p1Down, p1Left, p1Right);

            if (car2) {
                const p2Up = keys['ArrowUp'];
                const p2Down = keys['ArrowDown'];
                const p2Left = keys['ArrowLeft'];
                const p2Right = keys['ArrowRight'];
                simulateVehiclePhysics(car2, p2Up, p2Down, p2Left, p2Right);
            }

            if (skidmarks.length > 2000) skidmarks.shift();

            document.getElementById('sp').innerText = Math.round(Math.abs(car.speed) * 4.3);
            if (car2) {
                document.getElementById('sp2').innerText = Math.round(Math.abs(car2.speed) * 4.3);
            }

            if (gameActive && currentMode === 'split' && car2) {
                let dx = car.x - car2.x;
                let dy = car.y - car2.y;
                let dist = Math.sqrt(dx*dx + dy*dy);
                if (dist < 35) {
                    endGame(false); 
                }
            }

            if (gameActive && currentMode === 'chase') {
                cops.forEach((cop) => {
                    let target = car; 
                    const dx = target.x - cop.x, dy = target.y - cop.y;
                    const dist = Math.sqrt(dx*dx + dy*dy);
                    
                    cop.angle = Math.atan2(dy, dx); 
                    cop.maxSpeed = target.speed;
                    cop.speed = target.speed;
                    
                    cop.x += Math.cos(cop.angle) * cop.speed;
                    cop.y += Math.sin(cop.angle) * cop.speed;
                    cop.lightState = (cop.lightState + 1) % 16;

                    if (dist < 36) { endGame(false); }
                });
            }
        }

        function playMusic(genre) {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                musicGainNode = audioCtx.createGain();
                musicGainNode.gain.setValueAtTime(parseFloat(volSlider.value) * 0.2, audioCtx.currentTime);
                musicGainNode.connect(audioCtx.destination);
            }
            
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            
            if (synthInterval) clearInterval(synthInterval);

            document.querySelectorAll('.radio-btn').forEach(b => b.classList.remove('playing'));
            if (genre === 'off') return;

            const target = window.event?.target || document.querySelector(`[onclick="playMusic('` + genre + `')"]`);
            if(target) target.classList.add('playing');

            let index = 0, tempo = 120, notes = [60, 62, 64, 65, 67, 69, 71, 72];
            
            if (genre === 'pop') { tempo = 128; notes = [60, 64, 67, 64, 69, 65, 67, 62]; }
            else if (genre === 'rock') { tempo = 140; notes = [45, 45, 48, 45, 50, 48, 47, 43]; }
            else if (genre === 'calm') { tempo = 85; notes = [57, 60, 64, 62, 57, 60, 62, 59]; }

            synthInterval = setInterval(() => {
                if (!audioCtx || audioCtx.state === 'suspended') return;
                
                let time = audioCtx.currentTime;
                let osc = audioCtx.createOscillator();
                let noteGain = audioCtx.createGain();
                
                osc.connect(noteGain);
                noteGain.connect(musicGainNode);

                let base = notes[index % notes.length];
                if(genre == 'rock') {
                    osc.type = 'sawtooth'; osc.frequency.setValueAtTime(midiToFreq(base), time);
                    noteGain.gain.setValueAtTime(0.08, time);
                    noteGain.gain.exponentialRampToValueAtTime(0.001, time + 0.35);
                    osc.start(time); osc.stop(time + 0.35);
                } else if(genre == 'pop') {
                    osc.type = 'triangle'; osc.frequency.setValueAtTime(midiToFreq(base + 12), time);
                    noteGain.gain.setValueAtTime(0.12, time);
                    noteGain.gain.exponentialRampToValueAtTime(0.001, time + 0.25);
                    osc.start(time); osc.stop(time + 0.25);
                } else {
                    osc.type = 'sine'; osc.frequency.setValueAtTime(midiToFreq(base), time);
                    noteGain.gain.setValueAtTime(0.15, time);
                    noteGain.gain.exponentialRampToValueAtTime(0.001, time + 0.5);
                    osc.start(time); osc.stop(time + 0.5);
                }
                index++;
            }, (60 / tempo) * 1000);
        }

            const targetVehicle = prompt("ENTER EXACT CAR MODEL TO OVERRIDE:");
            if (targetVehicle !== null) {
                let lowerStr = targetVehicle.toLowerCase();
                let targetMatch = null;
                
                if (lowerStr === 'devxrbcar' || lowerStr === 'rb car' || lowerStr === 'rbcar') {
                    targetMatch = carCatalog.find(c => c.id === 'devxrbcar');
                    document.getElementById('p2-rbcar-opt').style.display = 'block';
                } else {
                    targetMatch = carCatalog.find(c => c.name.toLowerCase().includes(lowerStr) || c.id.toLowerCase().includes(lowerStr));
                }
                
                if (targetMatch) {
                    targetMatch.owned = true;
                    alert("code sucessfull"); 
                    
                    if (currentMode === 'split' && gameActive) {
                        activeCarId = targetMatch.id; 
                        activeCarProfile = targetMatch;
                        activeCarProfileP2 = targetMatch;
                        syncCarProfile(car, targetMatch);
                        if (car2) {
                            syncCarProfile(car2, targetMatch);
                        }
                        document.getElementById('p2-car-select').value = targetMatch.id;
                        localStorage.setItem('drift_active_car', targetMatch.id);
                    } else {
                        activeCarId = targetMatch.id; 
                        activeCarProfile = targetMatch;
                        syncCarProfile(car, targetMatch);
                        localStorage.setItem('drift_active_car', targetMatch.id);
                    }
                }
            }
        });
